cmake_minimum_required(VERSION 3.22)
project(djengine CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(ANDROID)
    find_package(oboe REQUIRED CONFIG)
    add_library(djengine SHARED engine.cpp)
    target_compile_options(djengine PRIVATE -O3 -fno-math-errno -DNDEBUG)
    target_link_libraries(djengine oboe::oboe log)
else()
    # Windows / macOS / Linux desktop build: djengine.dll / libdjengine.dylib / libdjengine.so  (miniaudio is bundled)
    add_library(djengine SHARED engine.cpp)
    set_target_properties(djengine PROPERTIES CXX_VISIBILITY_PRESET hidden)
    if(MSVC)
        target_compile_options(djengine PRIVATE /O2 /fp:fast /EHsc /utf-8)
        target_compile_definitions(djengine PRIVATE _CRT_SECURE_NO_WARNINGS)
    else()
        target_compile_options(djengine PRIVATE -O3 -fno-math-errno -DNDEBUG)
        find_package(Threads REQUIRED)
        target_link_libraries(djengine PRIVATE Threads::Threads ${CMAKE_DL_LIBS} m)
    endif()
endif()
